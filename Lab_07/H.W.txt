<?php

namespace App\Http\Controllers;

use App\Models\Book;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Auth;

class BookController extends Controller
{
    // show the dashboard with only the logged-in user's books
    public function index()
    {
        // newest books first, and only the ones this user added
        $books = Book::where('user_id', Auth::id())
                     ->latest()
                     ->get();

        return view('dashboard', compact('books'));
    }

    // validate the form and save a new book
    public function store(Request $request)
    {
        $validated = $request->validate([
            'title'  => 'required|string|max:150',
            'author' => 'required|string|max:100',
            'pages'  => 'required|integer|min:1|max:10000',
        ]);

        // the user id comes from the session, not from the form
        $validated['user_id'] = Auth::id();

        Book::create($validated);

        return redirect()->route('dashboard')
                         ->with('success', 'Book added successfully.');
    }
}
